% \newlabel is only valid while LaTeX is reading a .aux file, so the frozen
% cross-references cannot simply be \input: doing so raises "Can be used only in
% preamble" for every entry. Read them with \newlabel locally redefined to
% install the reference directly, which is what \ref and \cref look up.
%
% The \@ifundefined guard matters. LaTeX reads this document's own .aux at
% \begin{document}, so by the time this macro runs, labels the excerpt defines
% itself are already set. Without the guard the frozen copy would overwrite them
% with the MAIN document's numbering, and the excerpt would print, say, "Table
% VIII" for a table that is its own Table IV.
\makeatletter
\newcommand{\importfrozenlabels}[1]{%
  \begingroup
    \def\newlabel##1##2{%
      \@ifundefined{r@##1}{\global\@namedef{r@##1}{##2}}{}}%
    \input{#1}%
  \endgroup}
\makeatother

\newtheorem{definition}{Definition}
\newtheorem{theorem}{Theorem}

Reference material for co-authors. The prose here is deliberately
fragmentary and is not manuscript text. Read it for content and for numbers.
Do not paste it into the paper.}}
\end{center}

\importfrozenlabels{xrefs-frozen.tex}
